# Release ProGuard/R8 keep rules retaining the native JNI entry points;
# shrinking is currently disabled in app/build.gradle.
#

# openi2pd release rules.
#
# Release builds currently set minifyEnabled false (see app/build.gradle), so
# these rules are documentation for the distribution process rather than a
# claim that R8 shrinking has been validated. Do not add keep rules for the
# removed HTTP console or any JavaScript bridge: neither exists in this app.

# JNI entry points are resolved by their Java names. Keep them if minification
# is enabled later.
-keepclasseswithmembers,includedescriptorclasses class org.openi2pd.router.I2PD_JNI {
    native <methods>;
}

# The SAF bridge looks up these static methods by name from JNI
# (GetStaticMethodID), so obfuscation/removal would break document storage even
# though the class is not an entry point itself.
-keep class org.openi2pd.router.TorrentSafStorage { *; }
# The broker and its generated AIDL stubs are named by the manifest/interface.
-keep class org.openi2pd.router.TorrentSafStorageService { *; }
-keep interface org.openi2pd.router.ITorrentSafStorage { *; }
-keep class org.openi2pd.router.ITorrentSafStorage$* { *; }
